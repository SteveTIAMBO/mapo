// La collection "nova_directory" stocke les profils ADN des écoles
// auditées sur NOVA. MAPO peut les lire pour pré-remplir les
// paramètres école quand une école est déjà dans NOVA.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Règles existantes NOVA (NE PAS MODIFIER) ──
    // ... vos règles actuelles ...

    // ── Bridge NOVA ↔ MAPO ──

    // nova_directory : profils ADN des écoles auditées
    // Lecture publique (MAPO lit sans authentification NOVA)
    // Écriture : tout utilisateur NOVA authentifié
    match /nova_directory/{slug} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
